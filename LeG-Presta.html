    ctx.fillStyle = '#DAD7CE';

    ctx.fill();

    ctx.lineWidth = 1.3;

    ctx.strokeStyle = '#585650';

    ctx.stroke();


    ctx.beginPath();

    ctx.ellipse(cx, topY, topW/2 - 2, 3.2, 0, 0, Math.PI*2);

    ctx.fillStyle = '#5C5A54';

    ctx.fill();


    ctx.lineWidth = 1.3;

    ctx.strokeStyle = '#585650';

    ctx.fillStyle = '#D2CFC6';

    [-1, 1].forEach(side => {

      ctx.beginPath();

      ctx.ellipse(cx + side*(topW/2 + 7), topY + 7, 5, 6, 0, 0, Math.PI*2);

      ctx.fill();

      ctx.stroke();

    });


    ctx.restore();


    return { topY };

  }


  function draw(){

    ctx.clearRect(0,0,W,H);


    ctx.strokeStyle = 'rgba(43,38,32,0.15)';

    ctx.lineWidth = 2;

    ctx.beginPath();

    ctx.moveTo(0, H-14);

    ctx.lineTo(W, H-14);

    ctx.stroke();


    const { topY } = drawPot();


    if(caughtEmoji){

      const bounce = catchAnimT > 0 ? (catchAnimT/220) : 0;

      const liftY = bounce * 6;


      ctx.save();

      ctx.beginPath();

      ctx.rect(pan.x - pan.w, 0, pan.w*2, topY + 0.5);

      ctx.clip();


      ctx.font = '26px sans-serif';

      ctx.textAlign = 'center';

      ctx.textBaseline = 'middle';

      ctx.fillText(caughtEmoji, pan.x, topY - liftY);

      ctx.restore();

    }


    ctx.font = '24px sans-serif';

    ctx.textAlign = 'center';

    ctx.textBaseline = 'middle';

    for(const it of items){

      ctx.fillText(it.emoji, it.x, it.y);

    }

  }


  function pointerX(clientX){

    const rect = canvas.getBoundingClientRect();

    let x = clientX - rect.left;

    return Math.max(pan.w/2, Math.min(W - pan.w/2, x));

  }


  canvas.addEventListener('pointerdown', e => {

    if(!running) return;

    pan.x = pointerX(e.clientX);

  });

  canvas.addEventListener('pointermove', e => {

    if(!running) return;

    if(e.pressure === 0 && e.pointerType === 'mouse' && !e.buttons) return;

    pan.x = pointerX(e.clientX);

  });


  function startGame(){

    resizeCanvas();

    reset();

    running = true;

    gmStarted = true;

    startScreen.classList.add('gm-hidden');

    endScreen.classList.add('gm-hidden');

    ensureLoop();

  }


  function endGame(){

    running = false;

    finalScoreEl.textContent = 'Score : ' + score + ' pts';

    endScreen.classList.remove('gm-hidden');

  }


  document.getElementById('gmStartBtn').addEventListener('click', startGame);

  document.getElementById('gmRetryBtn').addEventListener('click', startGame);


  resizeCanvas();

  reset();

  ensureLoop();

})();
